Code Samples

curl -X POST https://apexsmmboosting.com/api/v2 \
  -H "Content-Type: application/json" \
  -d '{
    "action": "add",
    "service": 101,
    "url": "https://instagram.com/my_page",
    "quantity": 1000,
    "mode": "demo"
  }'
fetch('https://apexsmmboosting.com/api/v2', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    action: 'add',
    service: 101,
    url: 'https://instagram.com/my_page',
    quantity: 1000,
    mode: 'demo'
  })
})
.then(res => res.json())
.then(data => console.log(data));

payload = {
    "action": "add",

response = requests.post("https://apexsmmboosting.com/api/v2", json=payload)
print(response.json())
